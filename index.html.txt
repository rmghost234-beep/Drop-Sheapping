function renderSearch(q){
  q=q.trim().toLowerCase();
  if(!q){ sResults.innerHTML=`<div class="sr-empty"><div class="big">🔍</div>Type to search ${P.length} products across ${CATS.length} categories.<br><span style="font-size:12px">Tip: press Ctrl+K anytime</span></div>`; return; }
  const hits=P.filter(p=>p.name.toLowerCase().includes(q)||catName(p.cat).toLowerCase().includes(q));
  if(!hits.length){ sResults.innerHTML=`<div class="sr-empty"><div class="big">🫥</div>No matches for "<b style="color:var(--ink)">${q}</b>". Try "watch", "lamp" or "travel".</div>`; return; }
  sResults.innerHTML = hits.map((p,i)=>`
    <div class="sr-item" style="animation-delay:${i*40}ms" onclick="closeSearch();openQuickView(${p.id})">
      <div class="sr-thumb" style="background:linear-gradient(150deg,${p.g[0]},${p.g[1]})">${p.emoji}</div>
      <div><h4>${p.name}</h4><p>${catName(p.cat)} · ★ ${p.rating}</p></div>
      <span class="sr-price">${fmt(p.price)}</span>
    </div>`).join('');
}
